HashMap stores key-value pairs. Hash function: sum of char codes % capacity. Collision handled by Separate Chaining.

How it works

  • Hash Function: hash = (sum of char codes) % 8
  • Insert: Calculate bucket, add to chain. If key exists, update value.
  • Search: Go to bucket, check chain for key.
  • Delete: Remove from chain.
  • Collision: Multiple keys at same bucket stored as chain.
CODE
cap = 8
table = [[] for _ in range(cap)]

def hash_fn(key):
    return sum(ord(c) for c in key) % cap

def insert(key, val):
    idx = hash_fn(key)
    for i,(k,v) in enumerate(table[idx]):
        if k == key:
            table[idx][i] = (key,val)
            return
    table[idx].append((key,val))

def search(key):
    idx = hash_fn(key)
    for k,v in table[idx]:
        if k == key: return v
    return "Not found"

def delete_key(key):
    idx = hash_fn(key)
    for i,(k,v) in enumerate(table[idx]):
        if k == key:
            table[idx].pop(i)
            return True
    return False

def display():
    for i in range(cap):
        print(f"Bucket {i}: {table[i]}")

# Example
insert("Alice","25"); insert("Bob","30")
print(search("Alice"))
display()

#include <iostream>
#include <vector>
using namespace std;

const int CAP=8;
vector<pair<string,string>> table[CAP];

int hashFn(string key){
    int sum=0;
    for(char c: key) sum+=c;
    return sum % CAP;
}
void insert(string key,string val){
    int idx=hashFn(key);
    for(auto &p: table[idx])
        if(p.first==key){ p.second=val; return; }
    table[idx].push_back({key,val});
}
string search(string key){
    int idx=hashFn(key);
    for(auto &p: table[idx])
        if(p.first==key) return p.second;
    return "Not found";
}
bool removeKey(string key){
    int idx=hashFn(key);
    for(int i=0;i<table[idx].size();i++)
        if(table[idx][i].first==key){
            table[idx].erase(table[idx].begin()+i);
            return true;
        }
    return false;
}
void display(){
    for(int i=0;i<CAP;i++){
        cout<<"Bucket "<<i<<": ";
        for(auto &p: table[i]) cout<<"["<<p.first<<":"<<p.second<<"] ";
        cout<<endl;
    }
}
int main(){
    insert("Alice","25"); insert("Bob","30");
    cout<<search("Alice")<<endl;
    display();
}

import java.util.*;

class HashMap{
    static int CAP=8;
    static ArrayList<String[]> table[] = new ArrayList[CAP];
    static{
        for(int i=0;i<CAP;i++) table[i]=new ArrayList<>();
    }
    static int hashFn(String key){
        int sum=0;
        for(char c: key.toCharArray()) sum+=c;
        return sum % CAP;
    }
    static void insert(String key,String val){
        int idx=hashFn(key);
        for(String[] p: table[idx])
            if(p[0].equals(key)){ p[1]=val; return; }
        table[idx].add(new String[]{key,val});
    }
    static String search(String key){
        int idx=hashFn(key);
        for(String[] p: table[idx])
            if(p[0].equals(key)) return p[1];
        return "Not found";
    }
    static boolean removeKey(String key){
        int idx=hashFn(key);
        for(int i=0;i<table[idx].size();i++)
            if(table[idx].get(i)[0].equals(key)){
                table[idx].remove(i); return true;
            }
        return false;
    }
    static void display(){
        for(int i=0;i<CAP;i++)
            System.out.println("Bucket "+i+": "+table[i]);
    }
    public static void main(String args[]){
        insert("Alice","25"); insert("Bob","30");
        System.out.println(search("Alice"));
        display();
    }
}
