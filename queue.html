Priority Queue

Priority Queue: smaller number = higher priority. Example: priority 1 is served before priority 3.

Ready. Smaller priority number = higher priority.
Display is sorted by priority (highest priority at front)

How it works

  • Enqueue: Add element at rear
  • Dequeue: Remove element from front
  • Circular: Uses (rear+1)%capacity to wrap around
  • Priority: Insert keeps list sorted by priority, delete removes highest priority

queue = []
def enqueue(x): queue.append(x)
def dequeue():
    if not queue: return "Underflow"
    return queue.pop(0)
def front():
    return queue[0] if queue else "Empty"

# Circular Queue
cap = 6
arr = [None]*cap
front = rear = -1
size = 0
def cq_enqueue(x):
    global front, rear, size
    if size==cap: return "Full"
    if front==-1: front=rear=0
    else: rear=(rear+1)%cap
    arr[rear]=x; size+=1
def cq_dequeue():
    global front, rear, size
    if size==0: return "Empty"
    val=arr[front]; arr[front]=None
    if size==1: front=rear=-1
    else: front=(front+1)%cap
    size-=1; return val

# Priority Queue (smaller number = higher priority)
pq = []
def pq_insert(val, pri):
    pq.append((pri,val))
    pq.sort()  # sort by priority
def pq_delete():
    if not pq: return "Empty"
    return pq.pop(0)

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

// Simple Queue
class Queue{
    int arr[20], front=0, rear=-1, cap=8, cnt=0;
public:
    bool enqueue(int x){
        if(cnt==cap) return false;
        rear=(rear+1)%cap; arr[rear]=x; cnt++; return true;
    }
    int dequeue(){
        if(cnt==0) return -1;
        int v=arr[front]; front=(front+1)%cap; cnt--; return v;
    }
    int getFront(){ return cnt==0?-1:arr[front]; }
};

// Circular Queue
class CircularQueue{
    int arr[6], front=-1, rear=-1, cap=6, cnt=0;
public:
    bool enqueue(int x){
        if(cnt==cap) return false;
        if(front==-1) front=rear=0;
        else rear=(rear+1)%cap;
        arr[rear]=x; cnt++; return true;
    }
    int dequeue(){
        if(cnt==0) return -1;
        int v=arr[front];
        if(cnt==1) front=rear=-1;
        else front=(front+1)%cap;
        cnt--; return v;
    }
};

// Priority Queue: smaller number = higher priority
struct Item{ int val, pri; };
class PriorityQueue{
    vector<Item> pq;
public:
    void insert(int v,int p){
        pq.push_back({v,p});
        sort(pq.begin(), pq.end(), [](Item a,Item b){return a.pri<b.pri;});
    }
    Item remove(){
        Item t=pq[0]; pq.erase(pq.begin()); return t;
    }
};

}

// Circular Queue
class CircularQueue{
    int arr[]=new int[6];
    int front=-1, rear=-1, cap=6, cnt=0;
    boolean enqueue(int x){
        if(cnt==cap) return false;
        if(front==-1) front=rear=0;
        else rear=(rear+1)%cap;
        arr[rear]=x; cnt++; return true;
    }
    int dequeue(){
        if(cnt==0) return -1;
        int v=arr[front];
        if(cnt==1) front=rear=-1;
        else front=(front+1)%cap;
        cnt--; return v;
    }
}

// Priority Queue: smaller number = higher priority
class PriorityQueue{
    class Item{ int val, pri; Item(int v,int p){val=v; pri=p;} }
    ArrayList<Item> pq=new ArrayList<>();
    void insert(int v,int p){
        pq.add(new Item(v,p));
        pq.sort((a,b)->a.pri-b.pri);
    }
    Item remove(){
        return pq.remove(0);
    }
}

public class Main{
    public static void main(String args[]){
        MyQueue q=new MyQueue(); q.enqueue(10);
        CircularQueue cq=new CircularQueue(); cq.enqueue(5);
        PriorityQueue pq=new PriorityQueue();
        pq.insert(10,2); pq.insert(20,1);
    }
}
