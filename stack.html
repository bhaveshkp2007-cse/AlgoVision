How it works

  • Push: Add element at top (top++)
  • Pop: Remove element from top (top--)
  • Peek: See top element without removing
  • Postfix: Scan left to right, push operands, apply operator on two top values
  • Prefix: Scan right to left, same logic

# Stack using list
stack = []

def push(x):
    stack.append(x)

def pop():
    if not stack:
        return "Underflow"
    return stack.pop()

def peek():
    if not stack:
        return "Empty"
    return stack[-1]

def display():
    print(stack)

# Postfix evaluation
def evaluate_postfix(expr):
    s = []
    for token in expr.split():
        if token.lstrip('-').isdigit():
            s.append(int(token))
        else:
            b = s.pop(); a = s.pop()
            if token == '+': s.append(a+b)
            elif token == '-': s.append(a-b)
            elif token == '*': s.append(a*b)
            elif token == '/': s.append(a/b)
            elif token == '^': s.append(a**b)
    return s[-1] if s else None

# Prefix evaluation
def evaluate_prefix(expr):
    s = []
    for token in reversed(expr.split()):
        if token.lstrip('-').isdigit():
            s.append(int(token))
        else:
            a = s.pop(); b = s.pop()
            if token == '+': s.append(a+b)
            elif token == '-': s.append(a-b)
            elif token == '*': s.append(a*b)
            elif token == '/': s.append(a/b)
    return s[-1] if s else None

# Example
push(10); push(20)
print(peek())  # 20
print(evaluate_postfix("5 2 3 * +"))  # 11
print(evaluate_prefix("+ 5 * 2 3"))  # 11

#include <iostream>
#include <stack>
#include <sstream>
using namespace std;

class Stack {
    int arr[20], top, cap;
public:
    Stack(int c=8){ cap=c; top=-1; }
    bool push(int x){
        if(top==cap-1) return false;
        arr[++top]=x; return true;
    }
    int pop(){
        if(top==-1) return -1;
        return arr[top--];
    }
    int peek(){
        if(top==-1) return -1;
        return arr[top];
    }
    void display(){
        for(int i=top;i>=0;i--) cout<<arr[i]<<" ";
        cout<<endl;
    }
};

int evaluatePostfix(string expr){
    stack<int> s;
    stringstream ss(expr);
    string t;
    while(ss>>t){
        if(isdigit(t[0]) || (t[0]=='-' && t.size()>1))
            s.push(stoi(t));
        else{
            int b=s.top(); s.pop();
            int a=s.top(); s.pop();
            if(t=="+") s.push(a+b);
            else if(t=="-") s.push(a-b);
            else if(t=="*") s.push(a*b);
            else if(t=="/") s.push(a/b);
        }
    }
    return s.top();
}

int evaluatePrefix(string expr){
    stack<int> s;
    // split and process reverse
    vector<string> tokens;
    stringstream ss(expr);
    string t; while(ss>>t) tokens.push_back(t);
    for(int i=tokens.size()-1;i>=0;i--){
        t=tokens[i];
        if(isdigit(t[0]) || (t[0]=='-' && t.size()>1))
            s.push(stoi(t));
        else{
            int a=s.top(); s.pop();
            int b=s.top(); s.pop();
            if(t=="+") s.push(a+b);
            else if(t=="-") s.push(a-b);
            else if(t=="*") s.push(a*b);
            else if(t=="/") s.push(a/b);
        }
    }
    return s.top();
}

    Stack s;
    s.push(10); s.push(20);
    cout<<s.peek()<<endl;
    cout<<evaluatePostfix("5 2 3 * +")<<endl;
    cout<<evaluatePrefix("+ 5 * 2 3")<<endl;

import java.util.Stack;

class MyStack {
    int arr[] = new int[20];
    int top = -1, cap = 8;
    boolean push(int x){
        if(top==cap-1) return false;
        arr[++top]=x; return true;
    }
    int pop(){
        if(top==-1) return -1;
        return arr[top--];
    }
    int peek(){
        if(top==-1) return -1;
        return arr[top];
    }
    void display(){
        for(int i=top;i>=0;i--) System.out.print(arr[i]+" ");
        System.out.println();
    }
    static int evaluatePostfix(String expr){
        Stack<Integer> s = new Stack<>();
        for(String t: expr.split(" ")){
            if(t.matches("-?\\d+")) s.push(Integer.parseInt(t));
            else{
                int b=s.pop(), a=s.pop();
                if(t.equals("+")) s.push(a+b);
                else if(t.equals("-")) s.push(a-b);
                else if(t.equals("*")) s.push(a*b);
                else if(t.equals("/")) s.push(a/b);
            }
        }
        return s.peek();
    }
    static int evaluatePrefix(String expr){
        Stack<Integer> s = new Stack<>();
        String[] tok = expr.split(" ");
        for(int i=tok.length-1;i>=0;i--){
            String t=tok[i];
            if(t.matches("-?\\d+")) s.push(Integer.parseInt(t));
            else{
                int a=s.pop(), b=s.pop();
                if(t.equals("+")) s.push(a+b);
                else if(t.equals("-")) s.push(a-b);
                else if(t.equals("*")) s.push(a*b);
                else if(t.equals("/")) s.push(a/b);
            }
        }
        return s.peek();
    }
    public static void main(String args[]){
        MyStack s=new MyStack();
        s.push(10); s.push(20);
        System.out.println(s.peek());
        System.out.println(evaluatePostfix("5 2 3 * +"));
        System.out.println(evaluatePrefix("+ 5 * 2 3"));
    }
}
