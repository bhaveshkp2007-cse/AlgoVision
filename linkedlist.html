Linked List Visualizer (Singly)

Linked List is collection of nodes. Each node has data and pointer to next node.



Ready. Insert nodes to build linked list.

Linked List Display

Size: 0 | HEAD: NULL

How it works

  • Insert: Create new node, link pointers
  • Delete: Update pointer to skip node
  • Search: Traverse from HEAD one by one
  • Display: Traverse and show all nodes -> NULL
CODE
class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None

    def insert_begin(self, val):
        n = Node(val)
        n.next = self.head
        self.head = n

    def insert_end(self, val):
        n = Node(val)
        if not self.head:
            self.head = n; return
        t = self.head
        while t.next: t = t.next
        t.next = n

    def delete_begin(self):
        if not self.head: return "Empty"
        val = self.head.data
        self.head = self.head.next
        return val

    def search(self, val):
        t = self.head; pos = 0
        while t:
            if t.data == val: return pos
            t = t.next; pos += 1
        return -1

    def display(self):
        t = self.head
        while t:
            print(t.data, end=" -> ")
            t = t.next
        print("NULL")

# Example
ll = LinkedList()
ll.insert_end(10); ll.insert_end(20)
ll.insert_begin(5)
ll.display()  # 5 -> 10 -> 20 -> NULL
print(ll.search(10))  # 1

class LinkedList{

        n.next=head; head=n;
    }
    void insertEnd(int v){

        while(t.next!=null) t=t.next;
        t.next=n;
